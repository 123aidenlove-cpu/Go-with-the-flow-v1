-- Supabase Setup (Run in SQL Editor)

-- 1. Profiles Table Updates
ALTER TABLE profiles ADD COLUMN IF NOT EXISTS name text;
ALTER TABLE profiles ADD COLUMN IF NOT EXISTS phone text;
ALTER TABLE profiles ADD COLUMN IF NOT EXISTS avatar_data jsonb DEFAULT '{"type": "owl"}'::jsonb;
ALTER TABLE profiles ADD COLUMN IF NOT EXISTS league_status text DEFAULT 'Bronze';

-- 2. Link Game Progress and Practice Logs to individual Profiles (Multi-student households)
ALTER TABLE game_progress ADD COLUMN IF NOT EXISTS profile_id uuid REFERENCES profiles(id);
ALTER TABLE practice_logs ADD COLUMN IF NOT EXISTS profile_id uuid REFERENCES profiles(id);

-- 3. Adventure Alerts Table
CREATE TABLE IF NOT EXISTS adventure_alerts (
    id uuid DEFAULT uuid_generate_v4() PRIMARY KEY,
    creator_id uuid REFERENCES profiles(id), -- Teacher or System
    target_profile_id uuid REFERENCES profiles(id),
    title text NOT NULL,
    type text NOT NULL, -- 'teacher' (orange) or 'app' (blue)
    components jsonb NOT NULL, -- Array of up to 3 tasks
    quavits_reward integer DEFAULT 0,
    expires_at timestamp with time zone,
    created_at timestamp with time zone DEFAULT now()
);

-- 4. SMART Goals Table
CREATE TABLE IF NOT EXISTS smart_goals (
    id uuid DEFAULT uuid_generate_v4() PRIMARY KEY,
    profile_id uuid REFERENCES profiles(id),
    timeframe text, -- 'Weekly', 'Monthly'
    focus text, -- 'sessions', 'minutes'
    target_value integer,
    current_value integer DEFAULT 0,
    priority_piece text,
    created_at timestamp with time zone DEFAULT now()
);

-- 5. Custom Repertoire Table
CREATE TABLE IF NOT EXISTS custom_repertoire (
    id uuid DEFAULT uuid_generate_v4() PRIMARY KEY,
    teacher_id uuid REFERENCES profiles(id),
    title text NOT NULL,
    composer text,
    audio_url text, -- Up to 90s recording
    practice_powers jsonb, -- up to 9 powers with audio guide links
    target_notes jsonb, -- up to 3 tricky notes
    target_scale text, -- foundational scale
    created_at timestamp with time zone DEFAULT now()
);
